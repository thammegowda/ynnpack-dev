if(CMAKE_VERSION VERSION_LESS 3.17)
    message(FATAL_ERROR "YNNPACK_ENABLE_CUDA requires CMake 3.17 or newer for FindCUDAToolkit")
endif()
find_package(CUDAToolkit 13 REQUIRED)
add_library(ynnpack_cuda STATIC runtime.cu)
target_include_directories(ynnpack_cuda PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(ynnpack_cuda PUBLIC CUDA::cudart CUDA::cublas)
set_target_properties(ynnpack_cuda PROPERTIES CUDA_STANDARD 17 POSITION_INDEPENDENT_CODE ON)
add_executable(ynnpack_cuda_test test.cpp)
target_link_libraries(ynnpack_cuda_test PRIVATE ynnpack_cuda)
add_executable(ynnpack_cuda_projection_benchmark projection_benchmark.cpp)
target_link_libraries(ynnpack_cuda_projection_benchmark PRIVATE ynnpack_cuda)
# Both stay buildable by name, but only join the default build when their option is on.
if(NOT YNNPACK_BUILD_TESTS)
    set_target_properties(ynnpack_cuda_test PROPERTIES EXCLUDE_FROM_ALL ON)
endif()
if(NOT YNNPACK_BUILD_BENCHMARKS)
    set_target_properties(ynnpack_cuda_projection_benchmark PROPERTIES EXCLUDE_FROM_ALL ON)
endif()
if(YNNPACK_BUILD_TESTS)
    add_test(NAME ynnpack_cuda_test COMMAND ynnpack_cuda_test)
endif()

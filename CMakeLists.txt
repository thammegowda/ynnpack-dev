cmake_minimum_required(VERSION 3.15 FATAL_ERROR)

project(ynnpack_dev LANGUAGES C CXX)

if(NOT DEFINED Python3_FIND_VIRTUALENV)
	set(Python3_FIND_VIRTUALENV STANDARD)
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)

option(YNNPACK_BUILD_TESTS "Build YNNPACK unit tests" OFF)
option(YNNPACK_BUILD_BENCHMARKS "Build YNNPACK benchmarks" OFF)
option(YNNPACK_USE_SYSTEM_LIBS "Use installed dependencies where supported" OFF)
if(EMSCRIPTEN OR YNNPACK_TARGET_PROCESSOR STREQUAL "wasm")
	set(YNN_CPUINFO_DEFAULT OFF)
else()
	set(YNN_CPUINFO_DEFAULT ON)
endif()
option(YNN_ENABLE_CPUINFO "Enable cpuinfo" ${YNN_CPUINFO_DEFAULT})

if(YNNPACK_BUILD_TESTS)
	enable_testing()
endif()

set(YNNPACK_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")

function(ynnpack_require_submodule name path)
	if(NOT EXISTS "${YNNPACK_THIRD_PARTY_DIR}/${path}/CMakeLists.txt")
		message(FATAL_ERROR "${name} is missing. Run: git submodule update --init --recursive")
	endif()
endfunction()

if(NOT TARGET slinky_base AND NOT YNNPACK_USE_SYSTEM_LIBS)
	ynnpack_require_submodule("Slinky" "slinky")
	set(SLINKY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
	add_subdirectory(third_party/slinky EXCLUDE_FROM_ALL)
	target_include_directories(slinky_base PUBLIC "${YNNPACK_THIRD_PARTY_DIR}/slinky")
	target_include_directories(slinky_chrome_trace PUBLIC "${YNNPACK_THIRD_PARTY_DIR}/slinky")
endif()

if(YNN_ENABLE_CPUINFO AND NOT TARGET cpuinfo)
	ynnpack_require_submodule("cpuinfo" "cpuinfo")
	set(CPUINFO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
	set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
	set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
	set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
	set(CPUINFO_BUILD_PKG_CONFIG OFF CACHE BOOL "" FORCE)
	set(_YNNPACK_SKIP_INSTALL_RULES "${CMAKE_SKIP_INSTALL_RULES}")
	set(CMAKE_SKIP_INSTALL_RULES ON)
	add_subdirectory(third_party/cpuinfo EXCLUDE_FROM_ALL)
	set(CMAKE_SKIP_INSTALL_RULES "${_YNNPACK_SKIP_INSTALL_RULES}")
	unset(_YNNPACK_SKIP_INSTALL_RULES)
endif()

if((YNNPACK_BUILD_TESTS OR YNNPACK_BUILD_BENCHMARKS) AND NOT TARGET gtest)
	ynnpack_require_submodule("GoogleTest" "googletest")
	set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
	set(gtest_build_tests OFF CACHE BOOL "" FORCE)
	set(gtest_build_samples OFF CACHE BOOL "" FORCE)
	add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
endif()

if(YNNPACK_BUILD_BENCHMARKS AND NOT TARGET benchmark::benchmark)
	ynnpack_require_submodule("Google Benchmark" "benchmark")
	set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
	set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
	set(BENCHMARK_ENABLE_DOXYGEN OFF CACHE BOOL "" FORCE)
	set(BENCHMARK_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
	set(BENCHMARK_INSTALL_TOOLS OFF CACHE BOOL "" FORCE)
	set(BENCHMARK_ENABLE_GTEST_TESTS OFF CACHE BOOL "" FORCE)
	add_subdirectory(third_party/benchmark EXCLUDE_FROM_ALL)
endif()

add_subdirectory(ynnpack)
